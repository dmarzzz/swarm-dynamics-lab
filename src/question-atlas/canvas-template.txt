import {useState,useCanvasState,useHostTheme,Stack,Row,Grid,H1,H2,H3,Text,TextInput,TextArea,Select,Button,Link,Divider} from 'cursor/canvas';
const atlas=__ATLAS_DATA__;
type Review={status?:string;notes?:string;candidate_sha256?:string};
export default function QuestionAtlas(){
 const theme=useHostTheme();
 const [query,setQuery]=useState('');const [area,setArea]=useState('');const [kind,setKind]=useState('');const [filter,setFilter]=useState('');
 const [selected,setSelected]=useState(atlas.candidates[0].id);
 const [reviews,setReviews]=useCanvasState<Record<string,Review>>('swarm-question-review-v1',{});
 const [change,setChange]=useState('');
 const needsRecheck=(d:typeof atlas.candidates[number])=>{const r=reviews[d.id];return !!r&&(r.candidate_sha256?r.candidate_sha256!==d.candidate_sha256:d.change==='revised')};
 const [showExport,setShowExport]=useState(false);
 const candidates=atlas.candidates.filter(d=>(!change||(change==='recheck'?needsRecheck(d):d.change===change))&&(!area||d.area===area)&&(!kind||d.novelty===kind)&&(!filter||(reviews[d.id]?.status||'Unreviewed')===filter)&&(!query||JSON.stringify(d).toLowerCase().includes(query.toLowerCase())));
 const current=candidates.find(d=>d.id===selected)||candidates[0];
 const update=(id:string,change:Review)=>setReviews(old=>({...old,[id]:{...(!old[id]?{candidate_sha256:atlas.candidates.find(d=>d.id===id)?.candidate_sha256}:{}),...old[id],...change}}));
 const section=(title:string,body:string)=><Stack gap={5}><H3>{title}</H3><Text>{body}</Text></Stack>;
 return <Stack gap={16} style={{padding:22,color:theme.text.primary,background:theme.bg.editor}}>
 <H1>{atlas.candidates.length} research questions for human review</H1>
 <Text tone="secondary">All {Object.keys(atlas.topics).length} research areas. Update {atlas.version}: {atlas.changes.new.length} new, {atlas.changes.revised.length} revised. Every item is an unreviewed hunch, with a prediction, comparison, falsifier and source trail. Nothing has been selected or run.</Text>
 <Row gap={10} wrap><TextInput value={query} onChange={setQuery} placeholder="Search questions, tests, sources" style={{minWidth:240,flex:1}}/>
 <Select value={area} onChange={setArea} options={[{value:'',label:'All research areas'},...Object.entries(atlas.topics).map(([value,label])=>({value,label}))]}/>
 <Select value={kind} onChange={setKind} options={[{value:'',label:'All kinds'},...['replication','boundary-test','extension','measurement','speculative'].map(value=>({value,label:value}))]}/>
 <Select value={filter} onChange={setFilter} options={[{value:'',label:'All review states'},...['Unreviewed','Shortlist','Discuss','Park'].map(value=>({value,label:value}))]}/><Select value={change} onChange={setChange} options={[{value:'',label:'All versions'},{value:'new',label:'New in this update'},{value:'revised',label:'Revised in this update'},{value:'recheck',label:'Needs review again'}]}/></Row>
 <Row gap={12} wrap><Text size="small">{candidates.length} shown · {Object.values(reviews).filter(r=>r.status==='Shortlist').length} shortlisted</Text><Button onClick={()=>{setQuery('');setArea('');setKind('');setFilter('');setChange('')}}>Clear filters</Button><Button onClick={()=>setShowExport(!showExport)}>Review JSON</Button><Link href="https://github.com/dmarzzz/swarm-lab/blob/main/synthesis/research-question-atlas.md">Review guide</Link></Row>
 {showExport&&<Stack gap={6}><Text size="small">Copy this JSON to share your selections. These are local notes, not experiment approvals.</Text><pre style={{whiteSpace:'pre-wrap',maxHeight:220,overflow:'auto',background:theme.fill.tertiary,padding:12}}>{JSON.stringify({schema:'swarm-lab-question-review-v1',atlas_sha256:atlas.content_sha256,review:reviews},null,2)}</pre></Stack>}
 {current&&<Grid columns="minmax(210px, 1fr) minmax(300px, 2fr)" gap={24}>
 <Stack gap={0} style={{maxHeight:'72vh',overflow:'auto'}}>{candidates.map(d=><button key={d.id} onClick={()=>setSelected(d.id)} style={{textAlign:'left',padding:'12px 10px',cursor:'pointer',color:theme.text.primary,background:d.id===current.id?theme.fill.secondary:'transparent',border:0,borderBottom:`1px solid ${theme.stroke.tertiary}`,font:'inherit'}}><div style={{fontSize:12,color:theme.text.secondary}}>{d.id} · {reviews[d.id]?.status||'Unreviewed'} · {d.change}{needsRecheck(d)?' · Review again':''}</div><div style={{fontWeight:600,marginTop:4}}>{d.title}</div></button>)}</Stack>
 <Stack gap={17} style={{maxHeight:'72vh',overflow:'auto',paddingRight:8}}><Text size="small" tone="secondary">{current.id} · {current.area} · {current.novelty} · {current.feasibility}</Text><H2>{current.title}</H2>{needsRecheck(current)&&<Text>Updated candidate: your earlier choice and notes are preserved. Review the changed question and test before keeping your choice.</Text>}
 {section('Question',current.question)}{section('Candidate hypothesis',current.hypothesis)}{section('How we could test it',current.test)}{section('Comparison',current.baseline)}{section('Measurements',current.metrics.join('; '))}{section('Would count against it',current.falsifier)}{section('Confounds',current.confounds)}{section('Before promotion',current.needs)}
 <Divider/><H3>Closest prior work</H3>{current.prior.map(s=><Stack key={s.id} gap={4}><Link href={'https://github.com/dmarzzz/swarm-lab/blob/main/'+s.path}>{s.title}</Link><Text>{s.relation}</Text><Text size="small" tone="secondary">Catalogue depth: {s.catalogued_depth}. This pass does not upgrade that reading claim.</Text></Stack>)}
 <Divider/><H3>Your review</H3><Select value={reviews[current.id]?.status||'Unreviewed'} onChange={status=>update(current.id,{status,candidate_sha256:current.candidate_sha256})} options={['Unreviewed','Shortlist','Discuss','Park'].map(value=>({value,label:value}))}/><TextArea value={reviews[current.id]?.notes||''} onChange={notes=>update(current.id,{notes})} placeholder="Why pursue, discuss or park this?"/>{needsRecheck(current)&&<Button onClick={()=>update(current.id,{candidate_sha256:current.candidate_sha256})}>Keep my choice after reviewing this update</Button>}<Text size="small" tone="secondary">Saved in this local review. Source gates and independent human selection still apply.</Text>
 </Stack></Grid>}
 </Stack>;
}
